'use client';

import { useState, useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { supabase } from '@/lib/supabase';
import { fetchSchoolAndClassInfo } from '@/services/attendanceService';
import BottomNav from '@/components/BottomNav';

export default function TeacherDashboard() {
  const [loading, setLoading] = useState(true);
  const [profile, setProfile] = useState({
    name: 'Hafiz Setyo Pratomo',
    email: 'peacetokrates@gmail.com',
    nip: 'ID / NIP : 1988...',
    avatarUrl: '/tomothink-logo.png',
    tenantName: 'Ruang Mandiri Guru',
  });

  const [isRoleOpen, setIsRoleOpen] = useState(false);
  const [roles, setRoles] = useState([
    { id: '1', name: 'Guru Kelas / Pengajar Utama (All-in)', active: true },
  ]);

  const router = useRouter();

  useEffect(() => {
    async function loadUserData() {
      try {
        const { data: { session }, error } = await supabase.auth.getSession();
        if (error || !session || !session.user.email) {
          router.replace('/login');
          return;
        }

        const info = await fetchSchoolAndClassInfo(session.user.email);
        if (info) {
          setProfile({
            name: info.teacherName || 'Guru Pengajar',
            email: session.user.email,
            nip: 'ID / Tenant Active',
            avatarUrl: '/tomothink-logo.png',
            tenantName: info.schoolName,
          });
        }
      } catch (err) {
        console.error('Gagal memuat profil:', err);
      } finally {
        setLoading(false);
      }
    }

    loadUserData();
  }, [router]);

  const handleLogout = async () => {
    await supabase.auth.signOut();
    router.replace('/login');
  };

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-blue-100 font-sans">
        <p className="text-xs font-bold text-blue-800 animate-pulse">Memuat Beranda Kelasyik...</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-[#b5d1f2] text-slate-800 font-sans flex flex-col justify-between select-none">
      
      {/* KONTEN UTAMA */}
      <div className="max-w-md w-full mx-auto p-4 space-y-4 pb-28">
        
        {/* SECTION 1: HEADER PROFIL & MY ROLE */}
        <div className="bg-blue-600/90 backdrop-blur-md rounded-2xl p-4 shadow-lg text-white border border-blue-400/30 space-y-3 relative">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-3">
              <div className="w-16 h-16 rounded-xl overflow-hidden border-2 border-white/80 shadow-md bg-white shrink-0">
                <img 
                  src={profile.avatarUrl} 
                  alt="Profile" 
                  className="w-full h-full object-cover"
                  onError={(e)=>{(e.target as HTMLImageElement).src = 'https://via.placeholder.com/150'}}
                />
              </div>

              <div className="space-y-0.5">
                <h1 className="font-bold text-sm tracking-tight leading-tight">{profile.name}</h1>
                <p className="text-[11px] text-blue-100 opacity-90 truncate max-w-[160px]">{profile.email}</p>
                <span className="inline-block text-[10px] bg-blue-700/80 px-2 py-0.5 rounded font-mono border border-blue-400/40">
                  {profile.nip}
                </span>
              </div>
            </div>

            <button 
              onClick={handleLogout}
              title="Keluar Akun"
              className="w-12 h-12 rounded-2xl bg-cyan-400/80 hover:bg-cyan-300 flex items-center justify-center text-blue-900 shadow-md transition active:scale-95 cursor-pointer shrink-0"
            >
              <svg xmlns="http://www.w3.org/2000/svg" className="w-6 h-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2.5} d="M17 16l4-4m0 0l-4-4m4 4H7m6 4v1a3 3 0 01-3 3H6a3 3 0 01-3-3V7a3 3 0 013-3h4a3 3 0 013 3v1" />
              </svg>
            </button>
          </div>

          {/* TOMBOL MY ROLE & DROPDOWN (Posisi dropdown digeser ke samping/kanan tombol My Role) */}
          <div className="relative pt-1 border-t border-blue-500/40 flex items-center justify-between">
            <div className="relative inline-block">
              <button
                onClick={() => setIsRoleOpen(!isRoleOpen)}
                className="bg-blue-800 hover:bg-blue-900 text-white text-[10px] font-bold px-3 py-1 rounded-lg shadow border border-blue-400/50 flex items-center gap-1.5 transition cursor-pointer"
              >
                <span>My Role</span>
                <span className={`transition-transform duration-200 text-[9px] ${isRoleOpen ? 'rotate-90' : ''}`}>▶</span>
              </button>

              {isRoleOpen && (
                <div className="absolute left-full top-0 ml-2 w-56 bg-white rounded-xl shadow-2xl border border-slate-200 py-1.5 z-50 text-slate-700 space-y-1">
                  <div className="px-3 py-1 border-b border-slate-100 text-[9px] uppercase font-bold text-slate-400 tracking-wider">
                    Daftar Peran Aktif:
                  </div>
                  {roles.map((r, idx) => (
                    <div key={idx} className="px-3 py-1.5 text-xs font-semibold flex items-center justify-between bg-blue-50/50 text-blue-900">
                      <span>{r.name}</span>
                      <span className="text-[9px] bg-emerald-100 text-emerald-700 px-1.5 py-0.5 rounded font-bold">Aktif</span>
                    </div>
                  ))}
                </div>
              )}
            </div>
            
          </div>
        </div>

        {/* SECTION 2: MENU UTAMA */}
        <div className="space-y-3 pt-2">
          
          {/* Menu 1: Ruang Kelas */}
          <button
            onClick={() => router.push('/teacher/classes/manage')}
            className="w-full bg-gradient-to-r from-blue-600 to-blue-500 hover:from-blue-700 hover:to-blue-600 active:scale-[0.99] transition-all p-2.5 rounded-2xl shadow-md border border-blue-400/40 flex items-center justify-between group cursor-pointer"
          >
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 rounded-full border-2 border-white/80 bg-blue-700/60 flex items-center justify-center p-2 shadow-inner overflow-hidden shrink-0">
                <img src="/icon/classroom.png" alt="Ruang Kelas" className="w-full h-full object-contain filter drop-shadow" />
              </div>
              <span className="font-extrabold text-white text-sm tracking-wide uppercase">RUANG KELAS</span>
            </div>
            <div className="w-8 h-8 rounded-full bg-white/20 flex items-center justify-center text-white mr-1 group-hover:translate-x-0.5 transition-transform">
              <span className="text-sm font-bold">❯</span>
            </div>
          </button>

          {/* Menu 2: Daftar Guru */}
          <button
            onClick={() => alert('Fitur Daftar Guru (Segera Hadir)')}
            className="w-full bg-gradient-to-r from-blue-600 to-blue-500 hover:from-blue-700 hover:to-blue-600 active:scale-[0.99] transition-all p-2.5 rounded-2xl shadow-md border border-blue-400/40 flex items-center justify-between group cursor-pointer"
          >
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 rounded-full border-2 border-white/80 bg-blue-700/60 flex items-center justify-center p-2 shadow-inner overflow-hidden shrink-0">
                <img src="/icon/teacher.png" alt="Daftar Guru" className="w-full h-full object-contain filter drop-shadow" />
              </div>
              <span className="font-extrabold text-white text-sm tracking-wide uppercase">DAFTAR GURU</span>
            </div>
            <div className="w-8 h-8 rounded-full bg-white/20 flex items-center justify-center text-white mr-1 group-hover:translate-x-0.5 transition-transform">
              <span className="text-sm font-bold">❯</span>
            </div>
          </button>

          {/* Menu 3: Daftar Siswa */}
          <button
            onClick={() => router.push('/teacher/master-data')}
            className="w-full bg-gradient-to-r from-blue-600 to-blue-500 hover:from-blue-700 hover:to-blue-600 active:scale-[0.99] transition-all p-2.5 rounded-2xl shadow-md border border-blue-400/40 flex items-center justify-between group cursor-pointer"
          >
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 rounded-full border-2 border-white/80 bg-blue-700/60 flex items-center justify-center p-2 shadow-inner overflow-hidden shrink-0">
                <img src="/icon/students.png" alt="Daftar Siswa" className="w-full h-full object-contain filter drop-shadow" />
              </div>
              <span className="font-extrabold text-white text-sm tracking-wide uppercase">DAFTAR SISWA</span>
            </div>
            <div className="w-8 h-8 rounded-full bg-white/20 flex items-center justify-center text-white mr-1 group-hover:translate-x-0.5 transition-transform">
              <span className="text-sm font-bold">❯</span>
            </div>
          </button>

          {/* Menu 4: [DLL ....] (Canvas) */}
          <button
            onClick={() => router.push('/teacher/canvas')}
            className="w-full bg-gradient-to-r from-blue-600 to-blue-500 hover:from-blue-700 hover:to-blue-600 active:scale-[0.99] transition-all p-2.5 rounded-2xl shadow-md border border-blue-400/40 flex items-center justify-between group cursor-pointer"
          >
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 rounded-full border-2 border-white/80 bg-blue-700/60 flex items-center justify-center p-2.5 shadow-inner overflow-hidden shrink-0">
                <img src="/icon/puzzle_game.png" alt="Canvas" className="w-full h-full object-contain filter drop-shadow" onError={(e)=>{(e.target as HTMLImageElement).src = '/icon/classroom.png'}} />
              </div>
              <span className="font-extrabold text-white text-sm tracking-wide uppercase">[DLL ....] / Canvas</span>
            </div>
            <div className="w-8 h-8 rounded-full bg-white/20 flex items-center justify-center text-white mr-1 group-hover:translate-x-0.5 transition-transform">
              <span className="text-sm font-bold">❯</span>
            </div>
          </button>

        </div>

      </div>

      {/* BOTTOM NAV */}
      <BottomNav />
    </div>
  );
}
