//src/app/admin/classes/manage/page.tsx

'use client';

import { useState, useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { supabase } from '@/lib/supabase';
import BottomNav from '@/components/BottomNav';

export default function ManageClassesPage() {
  const router = useRouter();
  const [loading, setLoading] = useState(true);
  const [activeTab, setActiveTab] = useState<'mine' | 'all'>('mine');
  const [searchQuery, setSearchQuery] = useState('');
  const [classesList, setClassesList] = useState<any[]>([]);
  const [currentUserId, setCurrentUserId] = useState<string | null>(null);

  useEffect(() => {
    async function fetchClassesData() {
      try {
        setLoading(true);
        // 1. Dapatkan session user yang sedang login
        const { data: { session } } = await supabase.auth.getSession();
        if (!session || !session.user.email) {
          router.replace('/login');
          return;
        }

        // 2. Ambil data user profil untuk mendapatkan user_id & tenant_id
        const { data: userData, error: userError } = await supabase
          .from('users')
          .select('user_id, tenant_id')
          .eq('email', session.user.email)
          .single();

        if (userError || !userData) {
          console.error('Gagal mengambil data user:', userError);
          return;
        }

        setCurrentUserId(userData.user_id);

        // 3. Ambil data kelas berdasarkan tenant
        let query = supabase
          .from('classes')
          .select(`
            class_id,
            class_name,
            grade_level,
            capacity,
            homeroom_teacher_id,
            tenant_id
          `)
          .eq('tenant_id', userData.tenant_id);

        const { data: classesData, error: classError } = await query;

        if (classError) {
          console.error('Gagal memuat kelas:', classError);
        } else {
          // Ambil juga data relasi pengajar dari teacher_classes jika ada
          const { data: teacherClasses } = await supabase
            .from('teacher_classes')
            .select('class_id')
            .eq('teacher_id', userData.user_id);

          const assignedClassIds = teacherClasses?.map(tc => tc.class_id) || [];

          // Format data dengan tambahan status kepemilikan kelas untuk filter "Kelas Saya"
          const formatted = (classesData || []).map((cls) => {
            const isHomeroom = cls.homeroom_teacher_id === userData.user_id;
            const isAssigned = assignedClassIds.includes(cls.class_id);
            return {
              ...cls,
              isMyClass: isHomeroom || isAssigned,
              picText: isHomeroom ? 'Anda (Wali Kelas)' : (isAssigned ? 'Pengajar Mapel' : 'Guru Lain')
            };
          });

          setClassesList(formatted);
        }
      } catch (err) {
        console.error('Terjadi kesalahan:', err);
      } finally {
        setLoading(false);
      }
    }

    fetchClassesData();
  }, [router]);

  // Filter data berdasarkan Tab (Kelas Saya vs Semua) dan Search Query
  const filteredClasses = classesList.filter((cls) => {
    const matchesSearch = cls.class_name.toLowerCase().includes(searchQuery.toLowerCase()) ||
                          cls.grade_level?.toLowerCase().includes(searchQuery.toLowerCase());
    if (activeTab === 'mine') {
      return matchesSearch && cls.isMyClass;
    }
    return matchesSearch;
  });

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-[#b5d1f2] font-sans">
        <p className="text-xs font-bold text-blue-900 animate-pulse">Memuat Data Ruang Kelas...</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-[#b5d1f2] text-slate-800 font-sans flex flex-col justify-between select-none">
      
      {/* KONTEN UTAMA */}
      <div className="max-w-md w-full mx-auto p-4 space-y-4 pb-28">
        
        {/* HEADER JUDUL */}
        <div className="bg-blue-600/90 backdrop-blur-md rounded-2xl p-4 shadow-lg text-center border border-blue-400/30">
          <h1 className="font-extrabold text-white text-base tracking-wider uppercase">RUANG KELAS</h1>
          <p className="text-[11px] text-blue-100 mt-0.5">Kelola kelas binaan dan daftar pengampu</p>
        </div>

        {/* TOMBOL AKSI: + KELAS & TOGGLE TAB */}
        <div className="flex items-center gap-3">
          <button
            onClick={() => router.push('/teacher/classes/create')}
            className="flex-1 bg-cyan-400 hover:bg-cyan-300 active:scale-95 text-blue-950 font-extrabold text-xs py-2.5 px-4 rounded-xl shadow-md border border-cyan-200 transition flex items-center justify-center gap-1.5 cursor-pointer"
          >
            <span className="text-base leading-none">+</span>
            <span>Tambah Kelas</span>
          </button>

          <div className="flex bg-blue-900/40 p-1 rounded-xl border border-blue-400/30">
            <button
              onClick={() => setActiveTab('mine')}
              className={`px-3 py-1.5 text-xs font-bold rounded-lg transition cursor-pointer ${
                activeTab === 'mine' 
                  ? 'bg-blue-600 text-white shadow' 
                  : 'text-blue-900 hover:bg-blue-800/10'
              }`}
            >
              Kelas Saya
            </button>
            <button
              onClick={() => setActiveTab('all')}
              className={`px-3 py-1.5 text-xs font-bold rounded-lg transition cursor-pointer ${
                activeTab === 'all' 
                  ? 'bg-blue-600 text-white shadow' 
                  : 'text-blue-900 hover:bg-blue-800/10'
              }`}
            >
              Semua
            </button>
          </div>
        </div>

        {/* AREA FILTER & SEARCH */}
        <div className="bg-white/85 backdrop-blur-sm p-3 rounded-2xl shadow-md border border-blue-200 flex items-center gap-2">
          <svg xmlns="http://www.w3.org/2000/svg" className="w-4 h-4 text-slate-400 shrink-0 ml-1" fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
          </svg>
          <input
            type="text"
            placeholder="Cari nama kelas atau tingkat..."
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
            className="w-full bg-transparent text-xs text-slate-700 placeholder-slate-400 focus:outline-none"
          />
        </div>

        {/* TABEL DATA KELAS */}
        <div className="bg-white rounded-2xl shadow-lg border border-blue-200 overflow-hidden">
          <div className="overflow-x-auto">
            <table className="w-full text-left border-collapse">
              <thead>
                <tr className="bg-blue-600 text-white text-[11px] font-bold uppercase tracking-wider">
                  <th className="p-3">Nama Kelas</th>
                  <th className="p-3">Grade</th>
                  <th className="p-3">Kapasitas</th>
                  <th className="p-3">Status / PIC</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100 text-xs">
                {filteredClasses.length > 0 ? (
                  filteredClasses.map((cls, idx) => (
                    <tr 
                      key={cls.class_id} 
                      onClick={() => router.push(`/teacher/classes/${cls.class_id}`)}
                      className={`hover:bg-blue-50/80 transition cursor-pointer ${idx % 2 === 0 ? 'bg-white' : 'bg-slate-50/50'}`}
                    >
                      <td className="p-3 font-bold text-blue-900">{cls.class_name}</td>
                      <td className="p-3 text-slate-600">{cls.grade_level || '-'}</td>
                      <td className="p-3 text-slate-600">{cls.capacity ? `${cls.capacity} Siswa` : '-'}</td>
                      <td className="p-3">
                        <span className={`px-2 py-0.5 rounded text-[10px] font-bold ${
                          cls.isMyClass 
                            ? 'bg-emerald-100 text-emerald-800 border border-emerald-200' 
                            : 'bg-slate-100 text-slate-600'
                        }`}>
                          {cls.picText}
                        </span>
                      </td>
                    </tr>
                  ))
                ) : (
                  <tr>
                    <td colSpan={4} className="p-6 text-center text-slate-400 italic text-xs">
                      Tidak ada data kelas ditemukan.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>
        </div>

      </div>

      {/* BOTTOM NAV */}
      <BottomNav />
    </div>
  );
}
